def pick_driver():
    preferred = [
        "ODBC Driver 18 for SQL Server",
        "ODBC Driver 17 for SQL Server",
        "ODBC Driver 13 for SQL Server",
        "SQL Server Native Client 11.0",
        "SQL Server",
    ]
    installed = pyodbc.drivers()
    for d in preferred:
        if d in installed:
            return d
    raise RuntimeError(
        "No SQL Server ODBC driver found. "
        "Install 'ODBC Driver 18 for SQL Server' from Microsoft and try again."
    )


def db_connect(server, database="master"):
    driver = pick_driver()
    db = database.replace("}", "}}")
    cs = (
        f"DRIVER={{{driver}}};SERVER={server};DATABASE={{{db}}};"
        "Trusted_Connection=yes;"
    )
    if driver.startswith("ODBC Driver"):
        cs += "TrustServerCertificate=yes;"
    return pyodbc.connect(cs, timeout=10, autocommit=True)


def q(name):
    return "[" + str(name).replace("]", "]]") + "]"


def proc(name):
    return f"{q(ACCESS_DATABASE)}.{q(ACCESS_SCHEMA)}.{q(name)}"


def sql_msg(e):
    m = re.search(r"\[SQL Server\]([^\[]+?)\s*\(\d+\)", str(e))
    return m.group(1).strip() if m else str(e)


def fmt_value(v):
    if v is None:
        return ""
    if isinstance(v, datetime.datetime):
        return v.strftime(DATE_FORMAT)
    if isinstance(v, datetime.date):
        return v.strftime(DATE_FORMAT.split(" ")[0])
    if isinstance(v, decimal.Decimal):
        return format(v.normalize(), "f")
    return str(v)


def windows_user():
    user = os.environ.get("USERNAME") or getpass.getuser()
    dom = os.environ.get("USERDOMAIN", "")
    return f"{dom}\\{user}" if dom else user


def short_name(login):
    return str(login).split("\\")[-1]


def same_user(a, b):
    a, b = str(a).strip().lower(), str(b).strip().lower()
    if "\\" in a and "\\" in b:
        return a == b
    return short_name(a) == short_name(b)


def check_access(conn):
    row = conn.cursor().execute(f"EXEC {proc('usp_CheckAccess')}").fetchone()
    return bool(row[0]), bool(row[1])


def fetch_members(conn):
    cur = conn.cursor().execute(f"EXEC {proc('usp_ListMembers')}")
    return [tuple(r) for r in cur.fetchall()]


def save_member(conn, username, role):
    row = conn.cursor().execute(
        f"EXEC {proc('usp_SaveMember')} @Username = ?, @Role = ?", username, role
    ).fetchone()
    return str(row[0]) if row else "added"


def remove_member(conn, username):
    conn.cursor().execute(f"EXEC {proc('usp_RemoveMember')} @Username = ?", username).fetchall()


class AccessDenied(Exception):
    pass


class Session:
    def __init__(self, server, conn, login, role):
        self.server = server
        self.conn = conn
        self.login = login
        self.role = role

    @property
    def is_admin(self):
        return self.role == "Admin"

    def close(self):
        try:
            self.conn.close()
        except Exception:
            pass


def sign_in(server):
    conn = db_connect(server, "master")

    login = conn.cursor().execute("SELECT SUSER_SNAME()").fetchone()[0]
    try:
        is_member, is_admin = check_access(conn)
    except pyodbc.Error:
        conn.close()
        raise AccessDenied(
            f"Couldn't check your access in {ACCESS_DATABASE}. "
            "Ask a portal admin to check the setup."
        )
    if is_admin:
        return Session(server, conn, login, "Admin")
    if is_member:
        return Session(server, conn, login, "Member")
    conn.close()
    raise AccessDenied(
        f"{login} doesn't have access to this portal yet. Ask a portal admin to add you."
    )

def severity_color(text):
    key = str(text or "").strip().lower()
    for k, v in SEVERITY_COLORS.items():
        if k.lower() == key:

    return None

def mix(c1, c2, t):
    a, b = QColor(c1), QColor(c2)
    return QColor(
        int(a.red() + (b.red() - a.red()) * t),
        int(a.green() + (b.green() - a.green()) * t),
        int(a.blue() + (b.blue() - a.blue()) * t),
    )

def _dpr():
    s = QGuiApplication.primaryScreen()
    return max(2.0, s.devicePixelRatio() if s else 1.0)

        return QPointF(s * x, s * y)

AVATAR_COLORS = ["#00A3E0", "#7C3AED", "#0891B2", "#059669", "#D97706", "#DB2777"]


def initials(login):
    parts = [x for x in re.split(r"[.\s_\-]+", short_name(login)) if x]
    if not parts:
        return "?"
    return (parts[0][0] + (parts[1][0] if len(parts) > 1 else "")).upper()


def make_avatar(login, size=36):
    r = _dpr()
    pm = QPixmap(int(size * r), int(size * r))
    pm.setDevicePixelRatio(r)
    pm.fill(Qt.transparent)
    p = QPainter(pm)
    p.setRenderHint(QPainter.Antialiasing)
    color = AVATAR_COLORS[sum(map(ord, str(login).lower())) % len(AVATAR_COLORS)]
    p.setPen(Qt.NoPen)
    p.setBrush(QColor(color))
    p.drawEllipse(QRectF(0, 0, size, size))
    f = QFont("Segoe UI")
    f.setPixelSize(int(size * .38))
    f.setBold(True)
    p.setFont(f)
    p.setPen(QColor("white"))
    p.drawText(QRectF(0, 0, size, size), Qt.AlignCenter, initials(login))
    p.end()
    return pm

def brand_logo(width):
    h = round(width * 392 / 1024)
    lbl = QLabel()

        r = _dpr()
        pm = pm.scaled(int(width * r), int(h * r), Qt.KeepAspectRatio, Qt.SmoothTransformation)
        pm.setDevicePixelRatio(r)

    else:

        lbl.setText("Your logo here\n1024 × 392")
        lbl.setToolTip(f"Save your logo as: {path}")

VALUE_ROLE = Qt.UserRole
CHECK_ROLE = Qt.UserRole + 1


class DropdownDelegate(QStyledItemDelegate):
    def __init__(self, dd):
        super().__init__(dd)
        self.dd = dd

    def sizeHint(self, opt, idx):
        return QSize(opt.rect.width(), 38)

    def paint(self, p, opt, idx):
        p.save()
        p.setRenderHint(QPainter.Antialiasing)
        r = QRectF(opt.rect).adjusted(4, 1, -4, -1)
        checked = bool(idx.data(CHECK_ROLE))
        hover = bool(opt.state & QStyle.State_MouseOver)
        bg = "#E6F6FC" if checked else ("#F1F5F9" if hover else None)
        if bg:
            p.setPen(Qt.NoPen)
            p.setBrush(QColor(bg))
            p.drawRoundedRect(r, 6, 6)

        x = r.left() + 10
        if self.dd.multi:
            box = QRectF(x, r.center().y() - 9, 18, 18)
            if checked:
                p.setPen(Qt.NoPen)
                p.setBrush(QColor("#00A3E0"))
                p.drawRoundedRect(box, 4, 4)
                pen = QPen(QColor("white"), 2)
                pen.setCapStyle(Qt.RoundCap)
                pen.setJoinStyle(Qt.RoundJoin)
                p.setPen(pen)
                p.drawPolyline([QPointF(box.x() + 4.5, box.y() + 9.5),
                                QPointF(box.x() + 7.8, box.y() + 12.8),
                                QPointF(box.x() + 13.5, box.y() + 6)])
            else:
                p.setPen(QPen(QColor("#CBD5E1"), 1.5))
                p.setBrush(QColor("white"))
                p.drawRoundedRect(box, 4, 4)
            x += 28

        text = idx.data(Qt.DisplayRole) or ""
        color = severity_color(text) if self.dd.show_colors else None
        if color:
            p.setPen(Qt.NoPen)
            p.setBrush(QColor(color))
            p.drawEllipse(QRectF(x, r.center().y() - 4.5, 9, 9))
            x += 18

        right = r.right() - 10
        if not self.dd.multi and checked:
            p.drawPixmap(int(right - 16), int(r.center().y() - 8), make_pix("check", "#00A3E0", 16))
            right -= 24

        f = QFont(opt.font)
        if checked:
            f.setWeight(QFont.DemiBold)
        p.setFont(f)
        p.setPen(QColor("#0F172A"))
        tr = QRectF(x, r.top(), max(10, right - x), r.height())
        p.drawText(tr, Qt.AlignVCenter | Qt.AlignLeft,
                   p.fontMetrics().elidedText(text, Qt.ElideRight, int(tr.width())))
        p.restore()


class _Popup(QFrame):
    closed = Signal()

    def __init__(self, parent):
        super().__init__(parent, Qt.Popup | Qt.FramelessWindowHint | Qt.NoDropShadowWindowHint)
        self.setAttribute(Qt.WA_TranslucentBackground)

    def hideEvent(self, e):
        super().hideEvent(e)
        self.closed.emit()

            foot.addStretch(1)
            foot.addWidget(done)
            pl.addLayout(foot)

        outer.addWidget(panel)
        self.popup.closed.connect(self._on_closed)
        self._update_text()

    def set_options(self, values):
        if self.show_colors:
            new = sorted(values, key=lambda v: (SEV_RANK.get(fmt_value(v).strip().lower(), 99),
                                                fmt_value(v).lower()))
        else:
            new = sorted(values, key=lambda v: fmt_value(v).lower())
        old_sel = set(self.selected)
        self.selected &= set(new)
        if new == self.options:
            self._sync_checks()
        else:
            self.options = new
            self._rebuild()
        self._update_text()
        return self.selected != old_sel

    def value(self):
        return next(iter(self.selected), None)

    def selected_in_order(self):
        return [v for v in self.options if v in self.selected]

    def clear_selection(self):
        self.selected.clear()
        self._sync_checks()
        self._update_text()

    def _rebuild(self):
        sb = self.list.verticalScrollBar().value()
        self.list.clear()
        for i, v in enumerate(self.options):
            it = QListWidgetItem(fmt_value(v))
            it.setData(VALUE_ROLE, i)
            it.setData(CHECK_ROLE, v in self.selected)
            it.setFlags(Qt.ItemIsEnabled)
            self.list.addItem(it)
        self._apply_filter(self.search_edit.text(), keep_hint=True)
        self.list.verticalScrollBar().setValue(sb)

    def _sync_checks(self):
        for i in range(self.list.count()):
            it = self.list.item(i)
            it.setData(CHECK_ROLE, self.options[it.data(VALUE_ROLE)] in self.selected)

    def _is_list_input(self, text):
        t = text.strip()
        return self.multi and bool(t) and (bool(re.search(r"[,;]", t)) or len(t.split()) > 1)

    def _parse(self, text):
        by_text = {}
        for i, v in enumerate(self.options):
            by_text.setdefault(fmt_value(v).strip().lower(), i)

        def partial(word):
            hits = [i for i, v in enumerate(self.options) if word.lower() in fmt_value(v).lower()]
            return hits[0] if len(hits) == 1 else None

        def greedy(words):
            found, missing, i = [], [], 0
            while i < len(words):
                for j in range(len(words), i, -1):
                    phrase = " ".join(words[i:j]).lower()
                    if phrase in by_text:
                        found.append(by_text[phrase])
                        i = j
                        break
                else:
                    missing.append(words[i])
                    i += 1
            return found, missing

        found, missing = [], []
        for part in [p.strip() for p in re.split(r"[,;\t\r\n]+", text) if p.strip()]:
            if part.lower() in by_text:
                found.append(by_text[part.lower()])
                continue
            f, m = greedy(part.split())
            found += f
            for word in m:
                hit = partial(word) if len(word) > 1 else None
                if hit is not None:
                    found.append(hit)
                else:
                    missing.append(word)
        return list(dict.fromkeys(found)), list(dict.fromkeys(missing))

    @staticmethod
    def _missing_text(missing):
        return "  ·  Not found: " + ", ".join(missing[:5]) + ("…" if len(missing) > 5 else "")

    def _apply_filter(self, text, keep_hint=False):
        list_mode = self._is_list_input(text)
        found_set = set(self._parse(text)[0]) if list_mode else None
        t = text.strip().lower()
        visible = 0
        for i in range(self.list.count()):
            it = self.list.item(i)
            if list_mode:
                show = it.data(VALUE_ROLE) in found_set
            else:
                show = not t or t in it.text().lower()
            it.setHidden(not show)
            visible += show
        self.empty_lbl.setText("No matching items" if t else "No options available")
        self.empty_lbl.setVisible(visible == 0)
        self.list.setVisible(visible > 0)
        if list_mode:
            found, missing = self._parse(text)
            msg = f"Press Enter to select {len(found)} item(s)"
            if missing:
                msg += self._missing_text(missing)
            self._show_hint(msg, error=bool(missing) and not found)
        elif t and self.multi:
            self._show_hint("Press Enter to select the matching item")
        elif not keep_hint:
            self.hint.hide()
            self._fit_popup()

    def _show_hint(self, msg, error=False):
        self.hint.setText(msg)
        self.hint.setProperty("error", error)
        repolish(self.hint)
        self.hint.show()
        self._fit_popup()

    def _reset_search(self):
        self.search_edit.blockSignals(True)
        self.search_edit.clear()
        self.search_edit.blockSignals(False)
        self._apply_filter("", keep_hint=True)

    def _emit_changed(self):
        QTimer.singleShot(0, self.selectionChanged.emit)

    def _on_enter(self):
        text = self.search_edit.text()
        vis = [self.list.item(i) for i in range(self.list.count()) if not self.list.item(i).isHidden()]
        if not self.multi:
            if vis:
                self._on_click(vis[0])
            return
        if not text.strip():
            return
        found, missing = self._parse(text)
        if not found and len(vis) == 1:
            found, missing = [vis[0].data(VALUE_ROLE)], []
        if not found:
            msg = "No matching items" if not vis else "Several items match - type more, or click to pick"
            self._show_hint(msg, error=True)
            return
        for i in found:
            self.selected.add(self.options[i])
        self._reset_search()
        msg = f"Selected {len(found)} item(s)"
        if missing:
            msg += self._missing_text(missing)
        self._show_hint(msg, error=bool(missing))
        self._sync_checks()
        self._update_text()
        self._emit_changed()

    def _on_click(self, item):
        v = self.options[item.data(VALUE_ROLE)]
        if self.multi:
            if v in self.selected:
                self.selected.discard(v)
            else:
                self.selected.add(v)
            item.setData(CHECK_ROLE, v in self.selected)
            self._update_text()
            self._emit_changed()
        else:
            self.selected = {v}
            self._sync_checks()
            self._update_text()
            self.popup.hide()
            self._emit_changed()

    def _select_visible(self):
        for i in range(self.list.count()):
            it = self.list.item(i)
            if not it.isHidden():
                self.selected.add(self.options[it.data(VALUE_ROLE)])
        self._sync_checks()
        self._update_text()
        self._emit_changed()

    def _clear_clicked(self):
        self.clear_selection()
        self.hint.hide()
        self._emit_changed()

    def _update_text(self):
        sel = self.selected_in_order()
        if not sel:
            text = self.placeholder
            self.text_lbl.setProperty("placeholder", True)
            self.badge.hide()
        else:
            text = ", ".join(fmt_value(v) for v in sel)
            self.text_lbl.setProperty("placeholder", False)
            if self.multi and len(sel) > 1:
                self.badge.setText(str(len(sel)))
                self.badge.show()
            else:
                self.badge.hide()
        repolish(self.text_lbl)
        fm = self.text_lbl.fontMetrics()
        w = max(40, self.text_lbl.width())
        self.text_lbl.setText(fm.elidedText(text, Qt.ElideRight, w))
        self.text_lbl.setToolTip(text if sel else "")

    def resizeEvent(self, e):
        super().resizeEvent(e)
        self._update_text()

    def mousePressEvent(self, e):
        if e.button() == Qt.LeftButton:
            if time.monotonic() - self._last_close < 0.25:
                return
            self.open_popup()

    def keyPressEvent(self, e):
        if e.key() in (Qt.Key_Space, Qt.Key_Return, Qt.Key_Enter, Qt.Key_Down):
            self.open_popup()
        else:
            super().keyPressEvent(e)

    def _fit_popup(self):
        if self.popup.isVisible():
            self.popup.adjustSize()

    def open_popup(self):
        self.setProperty("open", True)
        repolish(self)
        rows = min(max(self.list.count(), 1), 7)
        self.list.setFixedHeight(rows * 38 + 6)
        self.popup.setFixedWidth(max(self.width(), 260))
        self.hint.hide()
        self.search_edit.clear()
        self._apply_filter("")
        self.popup.adjustSize()
        pos = self.mapToGlobal(QPoint(0, self.height() + 6))
        scr = self.screen().availableGeometry()
        if pos.y() + self.popup.height() > scr.bottom():
            pos = self.mapToGlobal(QPoint(0, -self.popup.height() - 6))
        self.popup.move(pos)
        self.popup.show()
        self.search_edit.setFocus()

    def _on_closed(self):
        self._last_close = time.monotonic()
        self.setProperty("open", False)
        repolish(self)

class SortItem(QTableWidgetItem):
    def __lt__(self, other):
        a, b = self.data(SORT_ROLE), other.data(SORT_ROLE)
        try:
            return a < b
        except TypeError:
            return self.text().lower() < other.text().lower()


def sort_key(v, is_sev):
    if v is None:
        return [1, ""]
    if is_sev:
        r = SEV_RANK.get(str(v).strip().lower())
        return [0, r if r is not None else 99, str(v).lower()]
    if isinstance(v, bool):
        return [0, int(v)]
    if isinstance(v, (int, float, decimal.Decimal)):
        return [0, float(v)]
    if isinstance(v, (datetime.date, datetime.datetime)):
        return [0, v.isoformat()]
    return [0, str(v).lower()]

def make_table(headers):
    t = QTableWidget(0, len(headers))
    t.setHorizontalHeaderLabels(headers)
    hh = t.horizontalHeader()
    hh.setSectionResizeMode(QHeaderView.Stretch)
    hh.setDefaultAlignment(Qt.AlignLeft | Qt.AlignVCenter)
    hh.setHighlightSections(False)
    hh.setFixedHeight(42)
    t.verticalHeader().setVisible(False)
    t.verticalHeader().setDefaultSectionSize(44)
    t.setEditTriggers(QAbstractItemView.NoEditTriggers)
    t.setSelectionBehavior(QAbstractItemView.SelectRows)
    t.setAlternatingRowColors(True)
    t.setShowGrid(False)
    t.setFocusPolicy(Qt.NoFocus)
    t.setWordWrap(False)
    return t

    def _sign_in(self):
        server = self.server_edit.text().strip()
        if not server:
            self.err1.show_msg("Enter the SQL Server name.")
            return
        self.err1.hide()
        self.signin_btn.setText("Signing in…")
        self.signin_btn.setEnabled(False)
        QApplication.processEvents()
        try:
            with busy():
                session = sign_in(server)
        except AccessDenied as e:
            self.err1.show_msg(str(e))
            return
        except Exception as e:
            detail = str(e)
            if len(detail) > 260:
                detail = detail[:260] + "…"
            self.err1.show_msg(f"Couldn't connect to {server}. Check the server name and "
                               f"your network/VPN.\n\n{detail}")
            return
        finally:
            self.signin_btn.setText("Sign in")
            self.signin_btn.setEnabled(True)

        self.session = session
        if self._load_databases():
            self._goto(1)

    def _load_databases(self):
        try:
            with busy():
                cur = self.session.conn.cursor()
                cur.execute(
                    "SELECT name FROM sys.databases "
                    "WHERE database_id > 4 AND state_desc = 'ONLINE' "
                    "AND HAS_DBACCESS(name) = 1 ORDER BY name"
                )
                dbs = [r[0] for r in cur.fetchall() if r[0].lower() != ACCESS_DATABASE.lower()]
        except Exception as e:
            self.err1.show_msg(f"Signed in, but couldn't list databases.\n\n{e}")
            return False
        self.db_dd.set_options(dbs)
        self.db_dd.clear_selection()
        self.tbl_dd.set_options([])
        self.tbl_dd.setEnabled(False)
        return True

    def _load_tables(self):
        db = self.db_dd.value()
        self.tbl_dd.set_options([])
        self.tbl_dd.setEnabled(False)
        if not db:
            return
        try:
            with busy():
                cur = self.session.conn.cursor()
                cur.execute(
                    f"SELECT TABLE_SCHEMA, TABLE_NAME, TABLE_TYPE "
                    f"FROM {q(db)}.INFORMATION_SCHEMA.TABLES ORDER BY TABLE_SCHEMA, TABLE_NAME"
                )
                rows = cur.fetchall()
        except Exception as e:
            self.err2.show_msg(f"Couldn't read tables from '{db}'.\n\n{e}")
            return
        self.err2.hide()
        self.table_map = {}
        for schema, name, ttype in rows:
            lbl = f"{schema}.{name}" + ("  (view)" if ttype == "VIEW" else "")
            self.table_map[lbl] = (schema, name)
        self.tbl_dd.set_options(list(self.table_map))
        self.tbl_dd.clear_selection()
        self.tbl_dd.setEnabled(True)

    def _open(self):
        db = self.db_dd.value()
        tlabel = self.tbl_dd.value()
        if not db or not tlabel:
            self.err2.show_msg("Select a database and a table.")
            return
        schema, table = self.table_map[tlabel]
        try:
            with busy():
                conn = db_connect(self.session.server, db)
                cur = conn.cursor()
                cur.execute(f"SELECT TOP 0 * FROM {q(schema)}.{q(table)}")
                cols = [d[0] for d in cur.description]
        except Exception as e:
            self.err2.show_msg(f"Couldn't open the table.\n\n{e}")
            return

        lower = {c.lower() for c in cols}
        needed = [c for c, _, _ in FILTERS] + [c for c, _ in RESULT_COLUMNS]
        if SEVERITY_COLUMN:
            needed.append(SEVERITY_COLUMN)
        missing = [c for c in dict.fromkeys(needed) if c.lower() not in lower]
        if missing:
            conn.close()
            self.err2.show_msg(
                "These columns from the CONFIG section aren't in this table: "
                + ", ".join(missing)
                + "\n\nColumns in this table: " + ", ".join(cols)
            )
            return
        self.err2.hide()
        self.opened.emit(self.session, conn, db, schema, table)

    def load_data(self):
        cols = ", ".join(q(c) for c, _, _ in FILTERS)
        try:
            with busy():
                cur = self.conn.cursor()
                cur.execute(f"SELECT DISTINCT {cols} FROM {self.table_sql}")
                self.rows = [tuple(r) for r in cur.fetchall()]
        except Exception as e:
            QMessageBox.critical(self, "Error", f"Couldn't load filter values.\n\n{e}")
            self.rows = []
        self.refresh_options()

    def refresh_options(self):
        n = len(self.dropdowns)
        for _ in range(n + 1):
            sel = [set(d.selected) for d in self.dropdowns]

            def match(row, skip):
                return all(i == skip or not sel[i] or row[i] in sel[i] for i in range(n))

            changed = False
            for i, dd in enumerate(self.dropdowns):
                values = {r[i] for r in self.rows if r[i] is not None and match(r, i)}
                if dd.set_options(values):
                    changed = True
            if not changed:
                break

    def run_search(self):
        cols = ", ".join(q(c) for c, _ in RESULT_COLUMNS)
        where, params = [], []
        for (col, _, _), dd in zip(FILTERS, self.dropdowns):
            vals = dd.selected_in_order()
            if vals:
                where.append(f"{q(col)} IN ({', '.join('?' * len(vals))})")
                params.extend(vals)
        sql = f"SELECT TOP ({MAX_RESULT_ROWS}) {cols} FROM {self.table_sql}"
        if where:
            sql += " WHERE " + " AND ".join(where)
        try:
            with busy():
                cur = self.conn.cursor()
                cur.execute(sql, params)
                data = cur.fetchall()
        except Exception as e:
            QMessageBox.critical(self, "Error", f"Search failed.\n\n{e}")
            return
        self.show_results(data)
        self._show_applied(len(data))

    def _show_applied(self, n):
        while self.chips_box.count():
            w = self.chips_box.takeAt(0).widget()
            if w:
                w.deleteLater()
        any_sel = False
        for (_, lbl, _), dd in zip(FILTERS, self.dropdowns):
            vals = [fmt_value(v) for v in dd.selected_in_order()]
            if not vals:
                continue
            any_sel = True
            shown = ", ".join(vals[:3]) + (f"  +{len(vals) - 3} more" if len(vals) > 3 else "")
            chip = QLabel(f"<b>{html.escape(lbl)}:</b> {html.escape(shown)}")
            chip.setObjectName("chip")
            chip.setToolTip(f"{lbl}: " + ", ".join(vals))
            self.chips_box.addWidget(chip)
        if not any_sel:
            chip = QLabel("None – showing all records")
            chip.setObjectName("chip")
            self.chips_box.addWidget(chip)
        limit = f" (limit {MAX_RESULT_ROWS:,})" if n >= MAX_RESULT_ROWS else ""
        self.applied_count.setText(f"{n:,} record{'s' if n != 1 else ''} found{limit}")
        self.applied.show()

    def show_results(self, data):
        self.table.setSortingEnabled(False)
        self.table.setRowCount(0)
        if not data:
            self.count_lbl.setText("0 records")
            self.empty_title.setText("No results to display")
            self.empty_sub.setText("Adjust your filters and try again.")
            self.stack.setCurrentIndex(0)
        else:
            self.table.setRowCount(len(data))
            for r, row in enumerate(data):
                for c, v in enumerate(row):
                    is_sev = c == self.sev_idx
                    item = SortItem(fmt_value(v))
                    item.setData(SORT_ROLE, sort_key(v, is_sev))
                    item.setToolTip(fmt_value(v))
                    self.table.setItem(r, c, item)
            self.table.setSortingEnabled(True)
            n = len(data)
            self.count_lbl.setText(f"{n:,} record{'s' if n != 1 else ''}")
            self.stack.setCurrentIndex(1)

    def clear_all(self):
        for dd in self.dropdowns:
            dd.clear_selection()
        self.refresh_options()
        self.table.setRowCount(0)
        self.count_lbl.setText("")
        self.applied.hide()
        self.empty_title.setText("No results to display")
        self.empty_sub.setText("Choose filters above and click Search.")
        self.stack.setCurrentIndex(0)

    def close_conn(self):
        try:
            self.conn.close()
        except Exception:
            pass

    def refresh(self):
        try:
            with busy():
                rows = fetch_members(self.session.conn)
        except Exception as e:
            self.table.setRowCount(0)
            self.count.setText("")
            self.msg.show_msg(f"Couldn't read the user list. Make sure portal_setup.sql has been "
                              f"run.\n\n{sql_msg(e)}")
            return
        self.table.setRowCount(len(rows))
        for r, (user, role, by, on) in enumerate(rows):
            role = "Admin" if str(role or "").strip().lower() == "admin" else "Member"
            self.table.setItem(r, 0, QTableWidgetItem(str(user)))
            badge = label(role, "roleBadge")
            badge.setProperty("role", role)
            self.table.setCellWidget(r, 1, self._wrap(badge))
            self.table.setItem(r, 2, QTableWidgetItem(str(by or "")))
            self.table.setItem(r, 3, QTableWidgetItem(fmt_value(on)))
            btn = QPushButton("  Remove")
            btn.setObjectName("danger")

            btn.setCursor(Qt.PointingHandCursor)
            btn.clicked.connect(lambda _=False, u=user: self.remove(u))
            self.table.setCellWidget(r, 4, self._wrap(btn))
        admins = sum(1 for _, role, *_ in rows if str(role or "").strip().lower() == "admin")
        self.count.setText(f"{len(rows)} user{'s' if len(rows) != 1 else ''}  ·  "
                           f"{admins} admin{'s' if admins != 1 else ''}")

    @staticmethod
    def _wrap(w):
        box = QWidget()
        l = QHBoxLayout(box)
        l.setContentsMargins(14, 0, 8, 0)
        box.setAttribute(Qt.WA_TranslucentBackground)
        if isinstance(w, QLabel):
            w.setFixedHeight(22)
            w.setFixedWidth(w.sizeHint().width() + 4)
        l.addWidget(w, 0, Qt.AlignVCenter)
        l.addStretch(1)
        return box

    def add(self):
        if not self.session.is_admin:
            return
        name = self.new_user.text().strip()
        role = self.role_combo.currentText()
        if not name:
            self.msg.show_msg("Enter a Windows username.")
            return
        if same_user(name, self.session.login) and role != "Admin":
            self.msg.show_msg("You can't remove your own admin rights.", "info")
            return
        try:
            with busy():
                result = save_member(self.session.conn, name, role)
        except Exception as e:
            self.msg.show_msg(f"Couldn't save {name}: {sql_msg(e)}")
            return
        tip = "" if "\\" in name else r"  Tip: include the domain (DOMAIN\username) to be exact."
        if result == "unchanged":
            self.msg.show_msg(f"{name} is already {'an' if role == 'Admin' else 'a'} {role}.", "info")
        elif result == "updated":
            self.msg.show_msg(f"{name} is now {'an' if role == 'Admin' else 'a'} {role}.", "success")
        else:
            self.msg.show_msg(f"Added {name} as {role}. They can sign in now.{tip}", "success")
        self.new_user.clear()
        self.refresh()

    def remove(self, username):
        if not self.session.is_admin:
            return
        if same_user(username, self.session.login):
            self.msg.show_msg("You can't remove yourself.", "info")
            return
        ans = QMessageBox.question(self, "Remove user",
                                   f"Remove {username} from the portal?\n"
                                   "They won't be able to sign in any more.")
        if ans != QMessageBox.Yes:
            return
        try:
            with busy():
                remove_member(self.session.conn, username)
        except Exception as e:
            self.msg.show_msg(f"Couldn't remove {username}: {sql_msg(e)}")
            return
        self.msg.show_msg(f"Removed {username}.", "success")
        self.refresh()

    def close_conn(self):
        self.dash.close_conn()


class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle(APP_NAME)
        self.setWindowIcon(QIcon(make_pix("logo", size=32)))
        self.resize(1360, 860)
        self.setMinimumSize(1100, 720)

        self.stack = QStackedWidget()
        self.stack.setObjectName("root")
        self.setCentralWidget(self.stack)
        self.login = LoginPage()
        self.login.opened.connect(self.open_portal)
        self.stack.addWidget(self.login)
        self.portal = None

    def open_portal(self, session, conn, db, schema, table):
        self._drop_portal()
        self.portal = PortalPage(session, conn, db, schema, table)
        self.portal.sign_out.connect(self.sign_out)
        self.portal.change_source.connect(self.change_source)
        self.stack.addWidget(self.portal)
        self.stack.setCurrentWidget(self.portal)

    def change_source(self):
        self._drop_portal()
        self.login.show_source_step()
        self.stack.setCurrentWidget(self.login)

    def sign_out(self):
        self._drop_portal()
        self.login.sign_out()
        self.stack.setCurrentWidget(self.login)

    def _drop_portal(self):
        if self.portal:
            self.portal.close_conn()
            self.stack.removeWidget(self.portal)
            self.portal.deleteLater()
            self.portal = None

    def closeEvent(self, e):
        self._drop_portal()
        self.login.sign_out()
        super().closeEvent(e)
