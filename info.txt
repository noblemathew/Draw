import struct
import sys

from PySide6.QtCore import QBuffer, QIODevice, Qt
from PySide6.QtWidgets import QApplication

import sql_filter_tool as app_code

qt = QApplication(sys.argv)
base = app_code.make_pix("logo", size=128).toImage()

sizes = [16, 24, 32, 48, 64, 128, 256]
images = []
for size in sizes:
    img = base.scaled(size, size, Qt.KeepAspectRatio, Qt.SmoothTransformation)
    buf = QBuffer()
    buf.open(QIODevice.WriteOnly)
    img.save(buf, "PNG")
    images.append(bytes(buf.data()))

header = struct.pack("<HHH", 0, 1, len(sizes))
offset = 6 + 16 * len(sizes)
entries = b""
data = b""
for size, png in zip(sizes, images):
    entries += struct.pack("<BBBBHHII", size % 256, size % 256, 0, 0, 1, 32, len(png), offset + len(data))
    data += png

with open("app.ico", "wb") as f:
    f.write(header + entries + data)

print("app.ico created")













def find_file(name):
    if os.path.isabs(name):
        return name if os.path.exists(name) else None
    places = [app_dir()]
    bundle = getattr(sys, "_MEIPASS", None)
    if bundle:
        places.append(bundle)
    for folder in places:
        path = os.path.join(folder, name)
        if os.path.exists(path):
            return path
    return None


def brand_logo(width, file=None):
    h = round(width * 392 / 1024)
    lbl = QLabel()
    lbl.setFixedSize(width, h)
    lbl.setAlignment(Qt.AlignCenter)
    path = find_file(file) if file else None
    if not path:
        path = find_file(LOGO_FILE)
    pm = QPixmap(path) if path else QPixmap()
    if not pm.isNull():
        r = _dpr()
        pm = pm.scaled(int(width * r), int(h * r), Qt.KeepAspectRatio, Qt.SmoothTransformation)
        pm.setDevicePixelRatio(r)
        lbl.setPixmap(pm)
    else:
        lbl.setObjectName("logoSlot")
        lbl.setText("Your logo here\n1024 × 392")
        lbl.setToolTip(f"Save your logo as: {os.path.join(app_dir(), file or LOGO_FILE)}")
    return lbl
